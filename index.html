} } function doGet(e) { try { const action = e.parameter.action || ''; let result; switch (action) { case 'loadQueue': result = loadQueue(); break; case 'loadConfig': result = loadConfig(); break; case 'ping': result = { success: true, time: new Date().toISOString() }; break; default: result = { success: true, status: 'Muse Sheets v7 running' }; } return jsonResponse(result); } catch(err) { logError('doGet', err); return jsonResponse({ success: false, error: err.message }); for (let i = 0; i < rows.length; i++) { if (rows[i][0] === today) { sheet.getRange(i + 2, 1, 1, 4).setValues([[today, now, device, json]]); return { success: true, action: 'updated', count: (data.queue||[]).length, updatedAt: now }; } } sheet.appendRow([today, now, device, json]); return { success: true, action: 'created', count: (data.queue||[]).length, updatedAt: now }; } function loadQueue() { const sheet = getQueueSheet(); const today = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd'); } } } return { success: true, queue: [], updatedAt: null }; } function clearQueue() { const sheet = getQueueSheet(); const today = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd'); } return { success: true, updatedAt: now }; } function loadConfig() { const sheet = getConfigSheet(); const lastRow = sheet.getLastRow(); if (lastRow < 2) return { success: true, config: null }; try { const row = sheet.getRange(2, 1, 1, 3).getValues()[0];
